Independent RGB-D visualization code; it does not participate in inference.

以下为独立 RGB-D 可视化代码，不参与推理。

import numpy as np
import matplotlib.pyplot as plt

# Visualize metric depth without changing the inference input.
# 将米制深度可视化，不修改供推理使用的 depth_m。
valid_depth = np.isfinite(depth_m) & (depth_m > 0)
depth_display_m = np.ma.masked_where(~valid_depth, depth_m)
depth_cmap = plt.get_cmap("viridis").with_extremes(bad="#202020")
fig, axes = plt.subplots(1, 2, figsize=(12, 5), layout="constrained")
axes[0].imshow(rgb)
axes[0].set_title("RGB")
depth_image = axes[1].imshow(depth_display_m, cmap=depth_cmap)
axes[1].set_title("Depth (black = invalid)")
for axis in axes:
    axis.set_axis_off()
# The colorbar uses the full valid depth range, in meters.
# 色条覆盖全部有效深度的范围，单位为米。
fig.colorbar(depth_image, ax=axes, location="bottom", shrink=0.6, label="Depth (m)")
plt.show()
